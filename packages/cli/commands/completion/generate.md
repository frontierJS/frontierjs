---
title: completion:generate
description: Print the shell completion script — pipe into your shell config to install
alias: cgen
examples:
  - fli completion:generate
  - fli completion:generate --shell zsh
  - fli completion:generate --shell bash
  - fli completion:generate --shell fish
flags:
  shell:
    char: s
    type: string
    description: "Shell to generate for (default: detected from $SHELL)"
    choices: [bash, zsh, fish]
    defaultValue: ''
---

<script>
const detectShell = () => {
  const shell = process.env.SHELL || ''
  if (shell.includes('zsh'))  return 'zsh'
  if (shell.includes('fish')) return 'fish'
  if (shell.includes('bash')) return 'bash'
  return 'bash'  // safe default
}

// ─── Bash ─────────────────────────────────────────────────────────────────────
// ':' is in COMP_WORDBREAKS, so COMP_WORDS holds "env:g" as "env" ":" "g". Filtering
// against COMP_WORDS[COMP_CWORD] matches "g" against "env:get" and finds nothing,
// so the word is taken from the raw line instead. Readline still replaces only
// the text after the last colon, so each reply loses that prefix or the line
// becomes "env:env:get". Bash cannot show a description, so each line is cut at
// its tab.
const bashScript = () => `# fli bash completion
# Generated by: fli completion:generate --shell bash
# Install:  eval "$(fli completion:generate --shell bash)"
# Or:       fli completion:generate --shell bash >> ~/.bash_completion

_fli_completion() {
  local line cur prefix answer value
  line="\${COMP_LINE:0:\$COMP_POINT}"
  cur="\${line##*[[:space:]]}"
  local IFS=$'\\n'
  local -a names=()
  answer="\$(fli completion:query "\$line" 2>/dev/null)"
  case "\$answer" in
    ${ANSWER.path}) COMPREPLY=(); compopt -o default 2>/dev/null; return ;;
    ${ANSWER.dir})  COMPREPLY=(); compopt -o dirnames 2>/dev/null; return ;;
  esac
  while IFS=$'\\t' read -r value _; do
    [[ -n "\$value" ]] && names+=("\$value")
  done <<< "\$answer"
  COMPREPLY=(\$(compgen -W "\${names[*]}" -- "\$cur"))
  if [[ "\$cur" == *:* && "\$COMP_WORDBREAKS" == *:* ]]; then
    prefix="\${cur%"\${cur##*:}"}"
    COMPREPLY=("\${COMPREPLY[@]#"\$prefix"}")
  fi
}
complete -F _fli_completion fli
`

// ─── Zsh ──────────────────────────────────────────────────────────────────────
// _describe reads "name:description", so a colon inside a name is escaped —
// otherwise "deploy:logs" shows as "deploy" described as "logs". Candidates
// sharing a description are grouped on one line, which is how an alias shows.
const zshScript = () => `# fli zsh completion
# Generated by: fli completion:generate --shell zsh
# Install:  eval "$(fli completion:generate --shell zsh)"
# Or:       fli completion:generate --shell zsh > "\${fpath[1]}/_fli"

_fli() {
  local line="\${(j: :)words[1,CURRENT]}"
  local -a answer candidates
  local entry name
  answer=("\${(@f)\$(fli completion:query "\$line" 2>/dev/null)}")
  case "\$answer" in
    ${ANSWER.path}) _files; return ;;
    ${ANSWER.dir})  _files -/; return ;;
  esac
  for entry in \$answer; do
    [[ -z "\$entry" ]] && continue
    name="\${entry%%$'\\t'*}"
    if [[ "\$entry" == *$'\\t'* ]]; then
      candidates+=("\${name//:/\\\\:}:\${entry#*$'\\t'}")
    else
      candidates+=("\${name//:/\\\\:}")
    fi
  done
  (( \${#candidates} )) && _describe -t fli 'fli' candidates
}
compdef _fli fli
`

// ─── Fish ─────────────────────────────────────────────────────────────────────
// Fish reads "value<tab>description" itself and has no colon problem. The
// token from commandline is unexpanded, and __fish_complete_path finds nothing
// under a literal ~, so it is expanded for the lookup and put back after.
const fishScript = () => `# fli fish completion
# Generated by: fli completion:generate --shell fish
# Install:  fli completion:generate --shell fish > ~/.config/fish/completions/fli.fish

# Disable default file completions for fli
complete -c fli -f

function __fli_paths -a lister
  set -l token (commandline -ct)
  if string match -q -- '~/*' $token
    $lister (string replace -r -- '^~' $HOME $token) | string replace -- $HOME '~'
  else
    $lister $token
  end
end

function __fli_complete
  set -l answer (fli completion:query (commandline -cp) 2>/dev/null)
  switch "$answer"
    case ${ANSWER.path}
      __fli_paths __fish_complete_path
    case ${ANSWER.dir}
      __fli_paths __fish_complete_directories
    case '*'
      printf '%s\\n' $answer
  end
end
complete -c fli -a "(__fli_complete)"
`
</script>

```js
const shell = flag.shell || detectShell()

switch (shell) {
  case 'zsh':
    process.stdout.write(zshScript())
    break
  case 'fish':
    process.stdout.write(fishScript())
    break
  case 'bash':
  default:
    process.stdout.write(bashScript())
    break
}
```
