// ── Environment / secrets health ──────────────────────────────────────────────
// Detects required secrets (parsed from .env.example, plus ENCRYPTION_KEY always)
// and checks whether each is actually set in the project .env / process.env.
// `fixable` secrets (ENCRYPTION_KEY, anything whose doc says "generate"/"openssl
// rand") can be auto-generated by the /fix/secret endpoint below.
const buildEnvHealth = (context) => {
  const root        = context.paths.root
  const envPath     = resolve(root, '.env')
  const examplePath = resolve(root, '.env.example')

  const known = []
  if (existsSync(examplePath)) {
    let comment = []
    for (const line of readFileSync(examplePath, 'utf8').split(/\r?\n/)) {
      const t = line.trim()
      if (t.startsWith('#')) { comment.push(t.replace(/^#+\s?/, '')); continue }
      const m = t.match(/^([A-Z][A-Z0-9_]*)\s*=/)
      if (m) {
        const desc     = comment.join(' ').trim()
        const required = /required/i.test(desc)
        const fixable  = /openssl rand|generate/i.test(desc) || m[1] === 'ENCRYPTION_KEY'
        known.push({ key: m[1], desc, required, fixable })
      }
      comment = []
    }
  }
  // ENCRYPTION_KEY is the one secret every FJS app needs — track it even if the
  // project has no .env.example yet.
  if (!known.some(k => k.key === 'ENCRYPTION_KEY')) {
    known.unshift({ key: 'ENCRYPTION_KEY', desc: 'App encryption key — generate with openssl rand -hex 32', required: true, fixable: true })
  }

  const fileVals = {}
  if (existsSync(envPath)) {
    for (const line of readFileSync(envPath, 'utf8').split(/\r?\n/)) {
      const m = line.match(/^\s*([A-Z][A-Z0-9_]*)\s*=\s*(.*)$/)
      if (m) fileVals[m[1]] = m[2].trim()
    }
  }
  const isSet = (key) => {
    const v = fileVals[key] !== undefined ? fileVals[key] : process.env[key]
    return v !== undefined && String(v).trim() !== ''
  }

  const vars = known.map(k => ({ ...k, status: isSet(k.key) ? 'set' : 'missing' }))
  return {
    file:            envPath,
    exampleExists:   existsSync(examplePath),
    vars,
    missingRequired: vars.filter(v => v.required && v.status === 'missing').length,
  }
}

// ── Map builder ───────────────────────────────────────────────────────────────

const buildMap = () => {
  try { freshJsonSchema(context) } catch (e) {
    log.warn(`Schema generation failed: ${e.message}`)
  }
  const schema = existsSync(resolve(context.paths.db, '.json', 'schema.json'))
    ? JSON.parse(readFileSync(resolve(context.paths.db, 'schema.json'), 'utf8'))
    : {}
  // Services come off the committed surface snapshot, never off a scan of
  // service source: what a service answers is decided at construction, not by
  // how a file reads, so a scanner drew a confident wrong picture with nothing
  // able to contradict it (FJS-254). No snapshot means no services panel — the
  // viewer is served to a person who can go and write one.
  const surface    = readApiSurface(context.paths.root)
  if (!surface) log.warn(surfaceMissingHint(context.paths.root))
  const resources  = scanFiles(context.paths.webResources, '.js', '.mesa')
                       .map(f => extractResourceMeta(readFileSync(f, 'utf8'), f))
  const migrations = parseMigrationFiles(resolve(context.paths.db, 'migrations'))
  const { packages } = extractServerMeta(context.paths.root)
  return {
    meta:       { generatedAt: new Date().toISOString(), root: context.paths.root },
    schema,
    services:   surface?.services.length ? surface.services : undefined,
    surface:    surface ? {
      file:        surface.file,
      generatedBy: surface.generatedBy,
      appHooks:    surface.appHooks,
      routes:      surface.routes,
      plugins:     surface.plugins,
    } : undefined,
    resources:  resources.length  ? resources  : undefined,
    migrations: migrations.length ? migrations : undefined,
    packages:   packages.length   ? packages   : undefined,
    env:        buildEnvHealth(context),
  }
}

// ── Injected env-health panel ─────────────────────────────────────────────────
// The viewer is a compiled React bundle (no JSX source shipped), so rather than
// surgically editing it we serve a small self-contained panel and inject a <script>
// tag at serve time. It reads the live `env` field from /data and offers a
// one-click fix (POST /fix/secret) for missing secrets. Tags are concatenated so
// the FLI compiler doesn't treat them as this command's own <script> block.
const PANEL_JS = `
(function () {
  var C = { bg:"#131411", border:"#2a2824", text:"#f0ede8", dim:"#6a6660", amber:"#d4920a", green:"#3fb950", red:"#e5534b", surface:"#0f0e0c" };
  var FONT = "ui-monospace, SFMono-Regular, Menlo, monospace";
  var panel, launcher, wrap, open = false, envData = null;

  function h(tag, style, text) {
    var e = document.createElement(tag);
    if (style) for (var k in style) e.style[k] = style[k];
    if (text != null) e.textContent = text;
    return e;
  }
  function badge(status) {
    var ok = status === "set";
    return h("span", { fontSize:"10px", padding:"1px 7px", borderRadius:"3px", fontWeight:"600",
      color: ok ? C.green : C.red, border:"1px solid " + (ok ? C.green : C.red) }, ok ? "set" : "missing");
  }
  function toast(msg, isErr) {
    var t = h("div", { position:"fixed", bottom:"60px", right:"20px", zIndex:"100001", maxWidth:"440px",
      background:C.bg, border:"1px solid " + (isErr ? C.red : C.amber), color:C.text, padding:"12px 14px",
      borderRadius:"8px", fontFamily:FONT, fontSize:"12px", lineHeight:"1.5", boxShadow:"0 8px 30px rgba(0,0,0,.5)" }, msg);
    document.body.appendChild(t);
    setTimeout(function () { t.remove(); }, 7000);
  }
  function fixSecret(key, btn) {
    btn.textContent = "working...";
    btn.disabled = true;
    fetch("/fix/secret", { method:"POST", headers:{ "Content-Type":"application/json" }, body: JSON.stringify({ key: key }) })
      .then(function (r) { return r.json(); })
      .then(function (res) {
        if (!res.ok) { btn.textContent = "failed"; toast("Could not fix " + key + ": " + (res.error || "error"), true); return; }
        toast(res.generated ? ("Generated " + key + " and wrote it to " + res.file) : ("Added " + key + "= to " + res.file + " — paste your value there"));
        load(true);
      })
      .catch(function () { btn.textContent = "error"; btn.disabled = false; });
  }
  function render() {
    if (!panel) return;
    panel.innerHTML = "";
    var head = h("div", { display:"flex", alignItems:"center", justifyContent:"space-between", marginBottom:"14px" });
    head.appendChild(h("div", { fontSize:"14px", fontWeight:"700", color:C.text, fontFamily:FONT }, "Environment health"));
    var x = h("div", { cursor:"pointer", color:C.dim, fontSize:"11px", fontFamily:FONT }, "close");
    x.onclick = function () { toggle(false); };
    head.appendChild(x);
    panel.appendChild(head);
    if (!envData || !envData.vars || !envData.vars.length) {
      panel.appendChild(h("div", { color:C.dim, fontSize:"12px", fontFamily:FONT }, "No environment data for this project."));
      return;
    }
    panel.appendChild(h("div", { fontSize:"11px", color:C.dim, fontFamily:FONT, marginBottom:"12px" }, "Secrets live in " + envData.file));
    envData.vars.forEach(function (v) {
      var row = h("div", { display:"flex", alignItems:"flex-start", gap:"10px", padding:"10px 12px", marginBottom:"8px",
        border:"1px solid " + C.border, borderRadius:"8px", background:C.surface });
      var left = h("div", { flex:"1", minWidth:"0" });
      var top = h("div", { display:"flex", alignItems:"center", gap:"8px", marginBottom:"3px", flexWrap:"wrap" });
      top.appendChild(h("span", { fontSize:"12px", fontWeight:"600", color:C.text, fontFamily:FONT }, v.key));
      if (v.required) top.appendChild(h("span", { fontSize:"9px", color:C.amber, fontFamily:FONT }, "required"));
      top.appendChild(badge(v.status));
      left.appendChild(top);
      if (v.desc) left.appendChild(h("div", { fontSize:"11px", color:C.dim, fontFamily:FONT, lineHeight:"1.4" }, v.desc));
      row.appendChild(left);
      if (v.status === "missing") {
        var btn = h("button", { background:"none", border:"1px solid " + C.amber, color:C.amber, borderRadius:"5px",
          padding:"6px 10px", fontSize:"11px", cursor:"pointer", fontFamily:FONT, whiteSpace:"nowrap", flexShrink:"0" },
          v.fixable ? "Generate & insert" : "Add to .env");
        btn.onclick = function () { fixSecret(v.key, btn); };
        row.appendChild(btn);
      }
      panel.appendChild(row);
    });
  }
  function updateLauncher() {
    if (!launcher) return;
    var miss = (envData && envData.missingRequired) ? envData.missingRequired : 0;
    launcher.textContent = miss > 0 ? ("Env health: " + miss + " missing") : "Env health: ok";
    launcher.style.borderColor = miss > 0 ? C.red : C.green;
    launcher.style.color = miss > 0 ? C.red : C.green;
  }
  function load(reopen) {
    fetch("/data").then(function (r) { return r.json(); }).then(function (m) {
      envData = m.env || null;
      updateLauncher();
      if (open || reopen) render();
    }).catch(function () {});
  }
  function toggle(force) {
    open = (force == null) ? !open : force;
    if (open) { render(); wrap.style.display = "block"; } else { wrap.style.display = "none"; }
  }

  wrap = h("div", { position:"fixed", top:"0", right:"0", height:"100%", width:"440px", maxWidth:"92vw", zIndex:"99999",

  panel = h("div", {});
  wrap.appendChild(panel);
  document.body.appendChild(wrap);

  launcher = h("div", { position:"fixed", top:"12px", right:"16px", zIndex:"99998", cursor:"pointer",
    background:C.bg, border:"1px solid " + C.green, color:C.green, borderRadius:"20px", padding:"5px 13px",
    fontSize:"11px", fontFamily:FONT, fontWeight:"600" }, "Env health: ...");
  launcher.onclick = function () { toggle(); };
  document.body.appendChild(launcher);

  load(false);
})();

const ENV_TAG = '<scr' + 'ipt src="/__envhealth.js"></scr' + 'ipt>'
const injectedHtml = !flag.legacy ? html
  : html.includes('</body>') ? html.replace('</body>', ENV_TAG + '</body>')
  : html + ENV_TAG

// The styling language and the highlighter, from the copy this fli holds —
// `core/assets.js` is the one owner, shared with the Web GUI and the generated
// map pages. Built once: the source cannot change under a running server.
const { ownStyleBundle, glowSource, CDN_STYLESHEET } = await import(resolve(global.fliRoot, 'core/assets.js'))
const stylesheet = ownStyleBundle()
const highlighter = glowSource()

const server = createServer(async (req, res) => {
  res.setHeader('Access-Control-Allow-Origin', '*')

  if (req.url === '/fli.css') {
    if (!stylesheet) { res.writeHead(302, { Location: CDN_STYLESHEET }); res.end(); return }
    res.writeHead(200, { 'Content-Type': 'text/css; charset=utf-8' })
    res.end(stylesheet)
    return
  }

  // A 404 is a legal answer: a code block renders unhighlighted rather than
  // the page failing to load.
  if (req.url === '/glow.js') {
    res.writeHead(highlighter ? 200 : 404, { 'Content-Type': 'text/javascript; charset=utf-8' })
    res.end(highlighter ?? '// @frontierjs/toolbelt not resolvable')
    return
  }

  if (req.url === '/data') {
    map = buildMap()
    res.writeHead(200, { 'Content-Type': 'application/json' })
    res.end(JSON.stringify(map))
    return
  }

  // ── /state ────────────────────────────────────────────────────────────────
  //
  // Whether the app this page maps is actually UP. The map is read off files
  // and says nothing about running processes, so a person could study a
  // complete chain of responsibility for an app that is not started — and the
  // page looked identical either way.
  //
  // Answered HERE rather than fetched from the app: a browser reaching
  // `localhost:8110/api/health` from this origin is a cross-origin request the
  // app has no reason to allow, and a CORS failure would read as *the app is
  // down*. The probe is `core/runnables.js`'s, the same one the fli GUI's
  // dashboard uses, so there is one answer to *is it answering* rather than
  // two that can disagree.
  if (req.url === '/state') {
    const { runnables, probeState } = await import(resolve(global.fliRoot, 'core/runnables.js'))
    // This app's own surfaces, and not the tooling block: `fli gui` being up
    // is not a fact about the app this page maps, and putting it in the badge
    // makes the badge mean two things.
    const rows  = runnables(context.paths.root).filter(r => r.kind === 'surface')
    const state = await probeState(rows)
    res.writeHead(200, { 'Content-Type': 'application/json' })
    res.end(JSON.stringify({
      at:   new Date().toISOString(),
      rows: rows.map(r => ({ id: r.id, name: r.name, dir: r.dir, port: r.port, open: r.open, start: r.start })),
      state,
    }))
    return
  }

  // Injected env-health panel script
  if (req.url === '/__envhealth.js') {
    res.writeHead(200, { 'Content-Type': 'application/javascript' })
    res.end(PANEL_JS)
    return
  }

  // One-click fix: generate/insert a managed secret into the project .env
  if (req.method === 'POST' && req.url === '/fix/secret') {
    let raw = ''
    req.on('data', c => raw += c)
    req.on('end', () => {
      let key
      try { key = JSON.parse(raw || '{}').key } catch {}
      const health = buildEnvHealth(context)
      const target = health.vars.find(v => v.key === key)
      if (!target) {
        res.writeHead(400, { 'Content-Type': 'application/json' })
        res.end(JSON.stringify({ ok: false, error: 'unknown or unmanaged key' }))
        return
      }
      const envPath = health.file
      const value   = target.fixable ? randomBytes(32).toString('hex') : ''
      let content   = existsSync(envPath) ? readFileSync(envPath, 'utf8') : ''
      const re      = new RegExp('^' + key + '\\s*=.*$', 'm')
      const lineStr = key + '=' + value
      if (re.test(content)) content = content.replace(re, lineStr)
      else content = content + (content && !content.endsWith('\n') ? '\n' : '') + lineStr + '\n'
      writeFileSync(envPath, content, 'utf8')
      if (value) process.env[key] = value  // reflect immediately so status flips to set
      log.success(`${key} ${value ? 'generated + written' : 'placeholder added'} → ${envPath}`)
      res.writeHead(200, { 'Content-Type': 'application/json' })
      res.end(JSON.stringify({ ok: true, key, file: envPath, value: value || null, generated: !!value }))
    })
    return
  }

  res.writeHead(200, { 'Content-Type': 'text/html' })
  res.end(injectedHtml)
})

server.listen(port, () => {
  log.success(`FJSChain viewer: ${url}${flag.legacy ? ' (legacy React viewer)' : ''}`)
  log.info('Refresh browser tab to update from current files')
  if (!flag.legacy) log.info('`--legacy` serves the previous viewer, for comparison')
})

// ── Open browser ───────────────────────────────────────────────────────────────

// minimist parses `--no-open` as { open: false }, so the flag is declared as
// `open` — declared as `no-open` it never bound, and the browser opened anyway.
if (flag.open) {
  const opener = {
    linux:  `xdg-open ${url}`,
    darwin: `open ${url}`,
    win32:  `start ${url}`,
  }[process.platform] || `xdg-open ${url}`

  setTimeout(() => {
    try { execSync(opener, { stdio: 'ignore' }) } catch {}
  }, 400)
}

log.info('Ctrl+C to stop')
await new Promise(() => {})
