<script>
import { existsSync, readFileSync, readdirSync, writeFileSync } from 'fs'
import { resolve, basename } from 'path'
import { execSync } from 'child_process'

// ─── freshJsonSchema ──────────────────────────────────────────────────────────
// Regenerates schema.json from schema.lite and returns the parsed object.
// Same approach as fli:validate — shells out to litestone jsonschema silently.

const freshJsonSchema = (context) => {
  const root = context.paths.root
  execSync(`cd ${root} && bunx litestone jsonschema --schema db/schema.lite`, {
    stdio: 'pipe',
  })
  const schemaJson = resolve(context.paths.db, 'schema.json')
  return JSON.parse(readFileSync(schemaJson, 'utf8'))
}

// ─── scanFiles ────────────────────────────────────────────────────────────────
// Returns all files under a directory matching one or more extensions.
// Recursive: an app may group a service in its own folder
// (api/src/services/apps/apps.service.ts) rather than keeping them flat.

const scanFiles = (dir, ...exts) => {
  if (!existsSync(dir)) return []
  const out = []
  const walk = (d) => {
    for (const e of readdirSync(d, { withFileTypes: true })) {
      const full = resolve(d, e.name)
      if (e.isDirectory()) {
        if (e.name === 'node_modules' || e.name.startsWith('.')) continue
        walk(full)
      } else if (e.isFile() && exts.some(x => e.name.endsWith(x))) {
        out.push(full)
      }
    }
  }
  walk(dir)
  return out.sort()
}

// ─── readApiSurface ───────────────────────────────────────────────────────────
//
// What the API answers, READ off the committed `surface.snapshot.md` rather than
// re-derived from service source.
//
// This replaces a scanner (`FJS-254`), and the reason is not that the scanner was
// buggy. It could not have been right: junction decides at CONSTRUCTION whether a
// key is an option or a method (`collectCustomMethods`, read back through
// `svc.describe()` — Invariant 4), `svc.pipelines()` resolves the hook chain, and
// `apiPrefix` moves every route. None of those answers is a fact about how a file
// reads. An action assigned from a module-level const takes no visible `ctx`, a
// `methods:` list built from a variable is not a literal to match, and a hook
// added by a plugin or by `app.hooks()` appears in no service file at all — so
// the scan drew a confident picture with nothing able to contradict it.
//
// `junction surface` writes the snapshot off a BUILT app and the `snapshots` CI
// phase fails a stale one, so what is read here is current or CI is red.
//
// **Absent means absent.** No snapshot returns null and the callers say so,
// naming the command; falling back to a scan is how a viewer ends up confidently
// wrong again, and a missing section is a question a person can answer.

// `core/app-entry.js` carries the same four pieces — this constant, the dirs
// below, the walk and the `generated by:` parse — because `app:atlas` reads the
// header off a file this parses. Whether the two commands merge is `FJS-D240`.
const SURFACE_FILE = 'surface.snapshot.md'

// Where it lives, in order. A snapshot belongs in the surface it describes —
// `db/` holds the Data realm's four, `web/` and `site/` hold their route
// tables, and the API's are facts about `api/`: its services, its routes, its
// hook chain, its plugins. `api/` is therefore the answer and is looked for
// first.
//
// The app root stays a candidate rather than being dropped, because the file
// is generated by a command an app already runs and an app that has not moved
// it is not broken — and because an app whose API is not in `api/` is a shape
// this cannot rule out. First hit wins, so an app holding both reads the one
// beside its API; two copies of this file is itself the defect
// `scripts/ci-allowances.json` records, and the `snapshots` phase rechecks
// each one against the directory it sits in.
const SURFACE_DIRS = ['api', '.']

const surfaceFile = (root) => {
  for (const dir of SURFACE_DIRS) {
    const file = resolve(root, dir, SURFACE_FILE)
    if (existsSync(file)) return file
  }
  return null
}

// `\`a\` → \`b\` → \`c\`` — a hook chain in run order. Also the bullet lists,
// which are the same backticked-and-comma-separated shape.
const backticked = (text) => [...text.matchAll(/`([^`]+)`/g)].map(m => m[1])

// | before | `find` | `gateAuth` → `autoFilter` |
const HOOK_ROW = /^\|\s*(before|after|around|error)\s*\|\s*`([^`]+)`\s*\|\s*(.*?)\s*\|\s*$/

const readApiSurface = (root) => {
  const file = surfaceFile(root)
  if (!file) return null

  const lines = readFileSync(file, 'utf8').split('\n')

  // The header names the command that wrote it, which is what a caller has to be
  // told to regenerate. Same line `fli test:snapshots` executes.
  const generatedBy = lines
    .find(l => l.includes('generated by:'))
    ?.replace(/^.*generated by:\s*/, '')
    .replace(/-->\s*$/, '')
    .trim() ?? null

  const services = []
  const appHooks = { before: {}, after: {}, around: {}, error: {} }
  const routes   = []
  const plugins  = []

  let section = null
  let svc     = null

  for (const line of lines) {
    if (line.startsWith('## ')) { section = line.slice(3).trim(); svc = null; continue }

    if (section === 'App hooks') {
      const m = line.match(HOOK_ROW)
      if (m) appHooks[m[1]][m[2]] = backticked(m[3])
      continue
    }

    if (section === 'Services') {
      // ### `alerts` · model `AlertRule`
      const head = line.match(/^###\s*`([^`]+)`(?:\s*·\s*model\s*`([^`]+)`)?/)
      if (head) {
        svc = {
          name:          head[1],
          model:         head[2] ?? head[1],
          methods:       [],
          customMethods: [],
          channel:       null,
          hooks:         { before: {}, after: {}, around: {}, error: {} },
        }
        services.push(svc)
        continue
      }
      if (!svc) continue

      const bullet = line.match(/^-\s*\*\*([^*]+)\*\*\s*—\s*(.*)$/)
      if (bullet) {
        const values = backticked(bullet[2])
        if (bullet[1] === 'methods')       svc.methods       = values
        if (bullet[1] === 'custom methods') svc.customMethods = values
        if (bullet[1] === 'broadcasts on') svc.channel       = values[0] ?? null
        continue
      }

      const hook = line.match(HOOK_ROW)
      if (hook) svc.hooks[hook[1]][hook[2]] = backticked(hook[3])
      continue
    }

    if (section === 'Routes') {
      // | GET | `/api/{service}` | service |
      const m = line.match(/^\|\s*([A-Z]+)\s*\|\s*`([^`]+)`\s*\|\s*(\w+)\s*\|\s*$/)
      if (m) routes.push({ method: m[1], path: m[2], kind: m[3] })
      continue
    }

    if (section === 'Plugins') {
      const m = line.match(/^\d+\.\s*`([^`]+)`/)
      if (m) plugins.push(m[1])
    }
  }

  return { file, generatedBy, services, appHooks, routes, plugins }
}

// The one sentence every caller says when there is no snapshot. It names the
// command rather than the file, because the file is not something anyone writes.
const surfaceMissingHint = (root) =>
  `no ${SURFACE_FILE} under ${root} (looked in ${SURFACE_DIRS.join(', ')}) — run ` +
  '`junction surface --app <entry>` from the API surface to write one. ' +
  'Services are read off a built app, never scanned (FJS-254).'

// ─── extractResourceMeta ──────────────────────────────────────────────────────
// Reads a resource module and extracts service, model and hook names.
//
// **Unlike a service, a Resource has no committed artefact to read.** It is
// constructed in the browser at runtime, so the only thing available is the
// `createResource(...)` call — and that call's arguments are literals, which is
// what makes reading them sound where scanning a service was not. What is read
// here is the call, not the file: everything below is matched INSIDE the
// argument list, and comments are stripped first, because a resource file is
// mostly prose and `model:` in a sentence is not a declaration (the same shape
// as the body tag written inside a comment, which this repo already has a rule
// for).
//
// The service name is the FIRST POSITIONAL ARGUMENT — it is not a `service:`
// key: `createResource('people', { model: 'Person' })`. The file name is the
// model where a model exists (Invariant 19), so a stated `model:` means the two
// disagree and is worth showing.

const stripComments = (src) =>
  src.replace(/\/\*[\s\S]*?\*\//g, ' ').replace(/(^|[^:])\/\/[^\n]*/g, '$1')

// The balanced argument list of the first createResource( … ) call, or null.
// Counted rather than matched: a nested `{ hooks: { before: { … } } }` ends three
// braces deep and `[^)]*` stops at the first one that closes an inner call.
const createResourceArgs = (src) => {
  const at = src.search(/createResource\s*\(/)
  if (at === -1) return null
  const open = src.indexOf('(', at)
  let depth = 0
  for (let i = open; i < src.length; i++) {
    if (src[i] === '(') depth++
    else if (src[i] === ')' && --depth === 0) return src.slice(open + 1, i)
  }
  return null   // unbalanced — an unparseable file says nothing rather than half
}

const extractResourceMeta = (src, filename) => {
  const name = basename(filename).replace(/\.(mesa|js)$/, '')
  const args = createResourceArgs(stripComments(src))
  if (!args) return { name, model: null, service: null, hooks: [] }

  const serviceM = args.match(/^\s*['"]([\w-]+)['"]/)
  const modelM   = args.match(/\bmodel\s*:\s*['"](\w+)['"]/)

  // hooks: { before: { create: [stampWorkspace] } } — read structurally, one
  // phase and one method at a time. The previous version swept the block for any
  // lowercase identifier minus a skip list, which counts words rather than
  // reading hooks: an argument, a property name and a hook all matched.
  const hookFns  = []
  const hooksAt  = args.match(/\bhooks\s*:\s*\{([\s\S]*)$/)
  if (hooksAt) {
    const phaseRe = /\b(before|after|around|error)\s*:\s*\{([^}]*)\}/g
    let pm
    while ((pm = phaseRe.exec(hooksAt[1])) !== null) {
      const methodRe = /\b\w+\s*:\s*\[([^\]]*)\]/g
      let mm
      while ((mm = methodRe.exec(pm[2])) !== null) {
        // A NAME, or nothing. An inline `ctx => stamp(ctx)` is anonymous, and
        // showing the text of it in a list of hook names is the same mistake at
        // one remove — the viewer's line reads as a name whatever is put there.
        for (const raw of mm[1].split(',')) {
          const fn = raw.trim()
          if (/^[A-Za-z_$][\w$]*(\.[A-Za-z_$][\w$]*)*$/.test(fn)) hookFns.push(fn)
        }
      }
    }
  }

  return {
    name,
    model:   modelM   ? modelM[1]   : null,
    service: serviceM ? serviceM[1] : null,
    hooks:   [...new Set(hookFns)],
  }
}

// ─── extractServerMeta ────────────────────────────────────────────────────────
// Reads server.ts (or server.js) and detects which tier-1 FJS packages are
// configured. Returns a structured packages array — one entry per known package,
// installed: true/false.
//
// Detection strategy: import presence + app.configure() call.
// Tier-1 packages: auth, conduit, caravan, notifications
// Tier-2 packages: litestream (config file), mailer (mailerPlugin call)

const TIER1_PACKAGES = [
  {
    id:    'auth',
    label: '@frontierjs/auth',
    short: 'auth',
    realm: 'junction',
    signals: ['createLitestoneAuth', 'createAuthPlugin', 'frontierjs/auth'],
  },
  {
    id:    'conduit',
    label: '@frontierjs/conduit',
    short: 'conduit',
    realm: 'junction',
    signals: ['conduit(', 'frontierjs/conduit', 'app.conduit'],
  },
  {
    id:    'caravan',
    label: '@frontierjs/caravan',
    short: 'caravan',
    realm: 'junction',
    signals: ['caravan(', 'frontierjs/caravan', 'app.caravan'],
  },
  {
    id:    'notifications',
    label: '@frontierjs/notifications',
    short: 'notifications',
    realm: 'junction',
    signals: ['notificationsPlugin', 'frontierjs/notifications', 'app.notify('],
  },
  {
    id:    'litestream',
    label: 'litestream',
    short: 'litestream',
    realm: 'litestone',
    signals: ['litestream', 'LITESTREAM'],
  },
]

const extractServerMeta = (root) => {
  // Candidate server entry files — check most specific first.
  // server.* is the scaffolded name; app.* and index.* are what hand-written
  // apps actually use (example/api/app.ts, basecamp/api/src/index.ts), and
  // missing them reported "no packages installed" for both.
  const names = ['server', 'app', 'index']
  const dirs  = ['api/src', 'api', '.']
  const candidates = dirs.flatMap(d =>
    names.flatMap(n => ['ts', 'js'].map(e => resolve(root, d, `${n}.${e}`)))
  )

  const serverFile = candidates.find(f => existsSync(f)) ?? null

  // Signals live wherever the app is COMPOSED, which is not always the entry:
  // basecamp's api/src/index.ts only calls buildBasecampApp(), and every
  // app.configure() sits in api/src/core/app.ts. Reading the entry alone
  // reported "no packages installed" for an app that configures four of them,
  // so scan the API tree.
  const apiDir  = resolve(root, 'api')
  const sources = existsSync(apiDir)
    ? scanFiles(apiDir, '.ts', '.js').filter(f => !/[\\/](test|tests|dist)[\\/]/.test(f))
    : []
  if (!serverFile && !sources.length) return { serverFile: null, packages: [] }

  const src = [
    serverFile ? readFileSync(serverFile, 'utf8') : '',
    ...sources.map(f => readFileSync(f, 'utf8')),
  ].join('\n')

  const packages = TIER1_PACKAGES.map(pkg => ({
    id:        pkg.id,
    label:     pkg.label,
    short:     pkg.short,
    realm:     pkg.realm,
    installed: pkg.signals.some(s => src.includes(s)),
  }))

  // Detect mailer separately (not in COR chips but useful metadata)
  const mailer =
    src.includes('createResendMailer') ? 'resend' :
    src.includes('createSmtpMailer')   ? 'smtp'   :
    src.includes('mailerPlugin')       ? 'unknown' :
    null

  return { serverFile, packages, mailer }
}
// Reads migration .sql files from db/migrations/ and returns structured rows.
// Does not query the database — file system only.

const parseMigrationFiles = (migrationsDir) => {
  if (!existsSync(migrationsDir)) return []
  return readdirSync(migrationsDir)
    // Both naming schemes in the wild: litestone's 14-digit timestamp
    // (20260804120000_add_orders.sql) and a hand-numbered 001_initial.sql
    .filter(f => /^\d+[_-][\w.-]*\.sql$/.test(f))
    .sort()
    .map(f => ({
      name: f,
      sql:  readFileSync(resolve(migrationsDir, f), 'utf8'),
    }))
}
// ── Environment / secrets health ──────────────────────────────────────────────
// Detects required secrets (parsed from .env.example, plus ENCRYPTION_KEY always)
// and checks whether each is actually set in the project .env / process.env.
// `fixable` secrets (ENCRYPTION_KEY, anything whose doc says "generate"/"openssl
// rand") can be auto-generated by the /fix/secret endpoint below.
const buildEnvHealth = (context) => {
  const root        = context.paths.root
  const envPath     = resolve(root, '.env')
  const examplePath = resolve(root, '.env.example')

  const known = []
  if (existsSync(examplePath)) {
    let comment = []
    for (const line of readFileSync(examplePath, 'utf8').split(/\r?\n/)) {
      const t = line.trim()
      if (t.startsWith('#')) { comment.push(t.replace(/^#+\s?/, '')); continue }
      const m = t.match(/^([A-Z][A-Z0-9_]*)\s*=/)
      if (m) {
        const desc     = comment.join(' ').trim()
        const required = /required/i.test(desc)
        const fixable  = /openssl rand|generate/i.test(desc) || m[1] === 'ENCRYPTION_KEY'
        known.push({ key: m[1], desc, required, fixable })
      }
      comment = []
    }
  }
  // ENCRYPTION_KEY is the one secret every FJS app needs — track it even if the
  // project has no .env.example yet.
  if (!known.some(k => k.key === 'ENCRYPTION_KEY')) {
    known.unshift({ key: 'ENCRYPTION_KEY', desc: 'App encryption key — generate with openssl rand -hex 32', required: true, fixable: true })
  }

  const fileVals = {}
  if (existsSync(envPath)) {
    for (const line of readFileSync(envPath, 'utf8').split(/\r?\n/)) {
      const m = line.match(/^\s*([A-Z][A-Z0-9_]*)\s*=\s*(.*)$/)
      if (m) fileVals[m[1]] = m[2].trim()
    }
  }
  const isSet = (key) => {
    const v = fileVals[key] !== undefined ? fileVals[key] : process.env[key]
    return v !== undefined && String(v).trim() !== ''
  }

  const vars = known.map(k => ({ ...k, status: isSet(k.key) ? 'set' : 'missing' }))
  return {
    file:            envPath,
    exampleExists:   existsSync(examplePath),
    vars,
    missingRequired: vars.filter(v => v.required && v.status === 'missing').length,
  }
}

// ─── buildProjectMap ──────────────────────────────────────────────────────────
//
// ONE reader of the project, for every presentation of it.
//
// There were two. `project:map` assembled its model inline and `project:view`
// had a `buildMap()` of its own, and the two answered differently about the same
// tree: 54 models against 42, because one counted by SHAPE and the fix for that
// landed only in the other (`FJS-1016`). They also collected different fields —
// env health and the app atlas existed in the page's model and in no other — so
// *what does this project contain* had two answers depending on which command
// you asked, which is `FJS-D223` one scope down: the split was a rendering
// choice wearing a command's clothes.
//
// `layer` narrows what is COLLECTED rather than what is shown, so a narrowed run
// is cheaper and its JSON says only what it looked at. `atlas` boots the app
// once for the three registers no file can answer; it degrades rather than
// fails, because a terminal map that needs bun would be a worse command than one
// that says why three sections are missing.

const buildProjectMap = async (context, { layer = '', atlas = true, log } = {}) => {
  const root          = context.paths.root
  const resourcesDir  = context.paths.webResources
  const migrationsDir = resolve(context.paths.db, 'migrations')
  const want          = (name) => !layer || layer === name
  const note          = (fn, msg) => { if (log) log[fn](msg) }

  let schema = null
  if (want('schema')) {
    try { schema = freshJsonSchema(context) }
    catch (e) { return { error: `litestone jsonschema failed: ${e.message}` } }
  }

  let packages = []
  if (want('api')) packages = extractServerMeta(root).packages

  // Read off the committed surface snapshot, never scanned from service source —
  // what a service ANSWERS is decided at construction and is not a fact about
  // how a file reads (FJS-254).
  let services = []
  let surface  = null
  if (want('api')) {
    surface = readApiSurface(root)
    if (surface) services = surface.services
    else note('warn', surfaceMissingHint(root))
  }

  let resources = []
  if (want('ui')) {
    resources = scanFiles(resourcesDir, '.js', '.mesa')
      .map(f => extractResourceMeta(readFileSync(f, 'utf8'), f))
  }

  let migrations = []
  if (want('migrations')) migrations = parseMigrationFiles(migrationsDir)

  // Three registers no file can answer — a job registers itself by being
  // autoloaded, a notification takes its type from its own file name, and a
  // principal resolver is installed in code. Read off a BUILT app, and DEGRADED
  // rather than fatal: reading a project must not require bun.
  let atlasPart = { skipped: true }
  if (atlas && want('api')) {
    const { readAppAtlas } = await import(new URL('file://' + global.fliRoot + '/core/app-entry.js'))
    const { model, error } = readAppAtlas(root)
    if (error) {
      note('warn', `Atlas unavailable, three sections will say why: ${error.split('\n')[0]}`)
      atlasPart = { error }
    } else {
      atlasPart = { jobs: model.jobs, notifications: model.notifications, principal: model.principal }
    }
  }

  const map = { meta: { generatedAt: new Date().toISOString(), root } }
  if (schema)            map.schema     = schema
  if (services.length)   map.services   = services
  if (resources.length)  map.resources  = resources
  if (migrations.length) map.migrations = migrations
  if (packages.length)   map.packages   = packages
  map.atlas = atlasPart
  map.env   = buildEnvHealth(context)

  // The rest of what the surface knows and nothing else here does: the hooks
  // that run around EVERY call, the paths the router actually mounted, and the
  // plugins in configure order. `packages` above is a signal scan over the API
  // tree and answers a different question — is this dependency present — so the
  // two are both carried rather than merged.
  if (surface) map.surface = {
    file:        surface.file,
    generatedBy: surface.generatedBy,
    appHooks:    surface.appHooks,
    routes:      surface.routes,
    plugins:     surface.plugins,
  }
  return { map }
}

// A model is the kind that SAYS model. `type === 'object'` is true of a `type`
// declaration and a view as well, which is how one presentation counted 54 and
// the other 42 over one tree (`FJS-1016`).
const countModels = (map) => Object.values(map.schema?.$defs ?? {})
  .filter(d => d['x-litestone-kind'] === 'model').length


// ─── serveProjectMap ──────────────────────────────────────────────────────────
//
// `--as=serve` — FJSChain over the same model the report and the JSON render.
// The one presentation here that does not exit: it is a server, and the flag
// description says so, because every other `--as` value anywhere answers and
// stops.

// `log` is a PARAMETER and not a global. A `<script>` block compiles to module
// scope, above `run()`, where `log` resolves to zx's — a function with no
// `.info` — so a helper that reads it throws `log.info is not a function` the
// first time it is called. The command body (a ```js fence) is inside `run()`
// and sees the real one, which is why this code worked in `project:view` and
// broke the moment it moved here. Everything this module needs is handed to it.
const serveProjectMap = async (context, initialMap, { port = 8501, open = true, atlas = true, log } = {}) => {
  const { createServer } = await import('http')
  const url  = `http://localhost:${port}`
  let   map  = initialMap

  const viewerPath = resolve(global.fliRoot, 'web/viewer/index.html')
  if (!existsSync(viewerPath)) {
    log.error(`Viewer not found at ${viewerPath}`)
    return
  }
  const html = readFileSync(viewerPath, 'utf8')

  // ── Start server ───────────────────────────────────────────────────────────────

  // The styling language and the highlighter, from the copy this fli holds —
  // `core/assets.js` is the one owner, shared with the Web GUI and the generated
  // map pages. Built once: the source cannot change under a running server.
  const { ownStyleBundle, glowSource, CDN_STYLESHEET } = await import(resolve(global.fliRoot, 'core/assets.js'))
  const stylesheet = ownStyleBundle()
  const highlighter = glowSource()

  const server = createServer(async (req, res) => {
    res.setHeader('Access-Control-Allow-Origin', '*')

    if (req.url === '/fli.css') {
      if (!stylesheet) { res.writeHead(302, { Location: CDN_STYLESHEET }); res.end(); return }
      res.writeHead(200, { 'Content-Type': 'text/css; charset=utf-8' })
      res.end(stylesheet)
      return
    }

    // A 404 is a legal answer: a code block renders unhighlighted rather than
    // the page failing to load.
    if (req.url === '/glow.js') {
      res.writeHead(highlighter ? 200 : 404, { 'Content-Type': 'text/javascript; charset=utf-8' })
      res.end(highlighter ?? '// @frontierjs/toolbelt not resolvable')
      return
    }

    // Re-read on every refresh, through the SAME builder the report and the JSON
    // use. It used to call a `buildMap` private to this command, which is how the
    // page and the terminal came to disagree about how many models there are.
    if (req.url === '/data') {
      const fresh = await buildProjectMap(context, { atlas, log: null })
      if (fresh.map) map = fresh.map
      res.writeHead(200, { 'Content-Type': 'application/json' })
      res.end(JSON.stringify(map))
      return
    }

    // ── /state ────────────────────────────────────────────────────────────────
    //
    // Whether the app this page maps is actually UP. The map is read off files
    // and says nothing about running processes, so a person could study a
    // complete chain of responsibility for an app that is not started — and the
    // page looked identical either way.
    //
    // Answered HERE rather than fetched from the app: a browser reaching
    // `localhost:8110/api/health` from this origin is a cross-origin request the
    // app has no reason to allow, and a CORS failure would read as *the app is
    // down*. The probe is `core/runnables.js`'s, the same one the fli GUI's
    // dashboard uses, so there is one answer to *is it answering* rather than
    // two that can disagree.
    if (req.url === '/state') {
      const { runnables, probeState } = await import(resolve(global.fliRoot, 'core/runnables.js'))
      // This app's own surfaces, and not the tooling block: `fli gui` being up
      // is not a fact about the app this page maps, and putting it in the badge
      // makes the badge mean two things.
      const rows  = runnables(context.paths.root).filter(r => r.kind === 'surface')
      const state = await probeState(rows)
      res.writeHead(200, { 'Content-Type': 'application/json' })
      res.end(JSON.stringify({
        at:   new Date().toISOString(),
        rows: rows.map(r => ({ id: r.id, name: r.name, dir: r.dir, port: r.port, open: r.open, start: r.start })),
        state,
      }))
      return
    }

    // One-click fix: generate/insert a managed secret into the project .env
    if (req.method === 'POST' && req.url === '/fix/secret') {
      let raw = ''
      req.on('data', c => raw += c)
      req.on('end', () => {
        let key
        try { key = JSON.parse(raw || '{}').key } catch {}
        const health = buildEnvHealth(context)
        const target = health.vars.find(v => v.key === key)
        if (!target) {
          res.writeHead(400, { 'Content-Type': 'application/json' })
          res.end(JSON.stringify({ ok: false, error: 'unknown or unmanaged key' }))
          return
        }
        const envPath = health.file
        const value   = target.fixable ? randomBytes(32).toString('hex') : ''
        let content   = existsSync(envPath) ? readFileSync(envPath, 'utf8') : ''
        const re      = new RegExp('^' + key + '\\s*=.*$', 'm')
        const lineStr = key + '=' + value
        if (re.test(content)) content = content.replace(re, lineStr)
        else content = content + (content && !content.endsWith('\n') ? '\n' : '') + lineStr + '\n'
        writeFileSync(envPath, content, 'utf8')
        if (value) process.env[key] = value  // reflect immediately so status flips to set
        log.success(`${key} ${value ? 'generated + written' : 'placeholder added'} → ${envPath}`)
        res.writeHead(200, { 'Content-Type': 'application/json' })
        res.end(JSON.stringify({ ok: true, key, file: envPath, value: value || null, generated: !!value }))
      })
      return
    }

    res.writeHead(200, { 'Content-Type': 'text/html' })
    res.end(html)
  })

  server.listen(port, () => {
    log.success(`FJSChain: ${url}`)
    log.info('Refresh browser tab to update from current files')
  })

  // ── Open browser ───────────────────────────────────────────────────────────────

  // minimist parses `--no-open` as { open: false }, so the flag is declared as
  // `open` — declared as `no-open` it never bound, and the browser opened anyway.
  if (open) {
    const opener = {
      linux:  `xdg-open ${url}`,
      darwin: `open ${url}`,
      win32:  `start ${url}`,
    }[process.platform] || `xdg-open ${url}`

    setTimeout(() => {
      try { execSync(opener, { stdio: 'ignore' }) } catch {}
    }, 400)
  }

  log.info('Ctrl+C to stop')
  await new Promise(() => {})
}

</script>
